export const interactionMaxAge = 3 * 60 * 1000
export function sanitizeInteractionAnchor(anchor){
  if(!anchor||!['before-answer','thinking','answering','after-answer','stopped'].includes(anchor.phase))return undefined
  const valid=(id,prefix)=>typeof id==='string'&&new RegExp('^'+prefix+'-[1-9][0-9]{0,5}$').test(id)
  return {phase:anchor.phase,...(valid(anchor.answerId,'answer')?{answerId:anchor.answerId}:{}),...(valid(anchor.userId,'user')?{userId:anchor.userId}: {})}
}

// Also used by the server: accept only known actions and exact local dialogue.
export function sanitizeInteractions(records, now = Date.now()) {
  if (!Array.isArray(records)) return []
  return records.slice(-5).flatMap(record => {
    if (!record || !Number.isFinite(record.at) || record.at > now || now-record.at > interactionMaxAge || !Array.isArray(record.steps)) return []
    const steps = record.steps.slice(-6).filter(step => step && Object.hasOwn(interactionLines, step.event) && interactionLines[step.event].includes(step.line)).map(({event,line,at}) => ({event,line,...(Number.isFinite(at)&&at<=now&&now-at<=interactionMaxAge?{at}:{})}))
    const anchor=sanitizeInteractionAnchor(record.anchor)
    return steps.length ? [{at:record.at,steps,...(anchor?{anchor}:{})}] : []
  })
}

export function createInteractionMemory(clock = Date.now) {
  let records = []
  return {
    add(event,line,anchor) {
      const now = clock()
      records = sanitizeInteractions(records,now)
      if (!Object.hasOwn(interactionLines,event) || !interactionLines[event].includes(line)) return
      const last = records.at(-1)
      const binding=sanitizeInteractionAnchor(anchor)
      const merge = last && JSON.stringify(last.anchor)===JSON.stringify(binding) && now-last.at < 10000 && last.steps.at(-1).event !== 'land' && event !== 'grab'
      if (merge) { last.at=now; last.steps.push({event,line,at:now}); last.steps=last.steps.slice(-6) }
      else records.push({at:now,steps:[{event,line,at:now}],...(binding?{anchor:binding}:{})})
      records=records.slice(-5)
    },
    snapshot() { records=sanitizeInteractions(records,clock()); return records.map(r=>({...r,...(r.anchor?{anchor:{...r.anchor}}:{}),steps:r.steps.map(s=>({...s}))})) },
    clear() { records=[] },
  }
}

export function createInteractionPicker(random = Math.random) {
  const previous = new Map()
  return event => {
    const pool = Object.hasOwn(interactionLines, event) ? interactionLines[event] : null
    if (!pool) return ''
    const choices = pool.filter(line => line !== previous.get(event))
    const line = choices[Math.floor(random() * choices.length)]
    previous.set(event, line)
    return line
  }
}
