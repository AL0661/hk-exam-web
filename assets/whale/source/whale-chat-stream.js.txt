// Read SSE across arbitrary byte boundaries, including split UTF-8 characters.
export async function readChatEvents(body,onEvent,signal){
  if(!body)throw new Error('聊天服务没有返回数据流。')
  const reader=body.getReader(),decoder=new TextDecoder();let buffer='',finished=false
  const consume=()=>{
    let end
    while((end=buffer.indexOf('\n\n'))>=0){
      const block=buffer.slice(0,end);buffer=buffer.slice(end+2)
      const data=block.split('\n').filter(line=>line.startsWith('data:')).map(line=>line.slice(5).trimStart()).join('\n')
      if(!data)continue
      if(data==='[DONE]'){finished=true;return}
      const event=JSON.parse(data)
      if(event.error)throw new Error(typeof event.error==='string'?event.error:event.error.message||'聊天服务返回错误。')
      onEvent(event)
    }
  }
  try{
    while(!finished){signal?.throwIfAborted();const {value,done}=await reader.read();buffer=(buffer+decoder.decode(value,{stream:!done})).replace(/\r\n/g,'\n');consume();if(done)break}
    if(!finished)throw new Error('连接中断，回答未完成，请重试。')
  }finally{await reader.cancel().catch(()=>{});reader.releaseLock()}
}
