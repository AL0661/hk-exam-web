export const normalizeMaterialQuery = normalizeSearch

const orderedEntries=ids=>ids.map(id=>verifiedMaterialEntries.find(r=>r.key===id)).filter(Boolean)

// Named textbooks take precedence over stage-level suggestions.
export function recommendedMaterialResults(message) {
  const q=normalizeMaterialQuery(message)
  const explicit=verifiedMaterialEntries.filter(r=>r.aliases?.some(alias=>q.includes(normalizeMaterialQuery(alias))))
  if(explicit.length)return explicit.sort((a,b)=>Math.max(...b.aliases.map(x=>normalizeMaterialQuery(x).length))-Math.max(...a.aliases.map(x=>normalizeMaterialQuery(x).length)))
  const primary=/小学|primary|pyp|ks[12]/.test(q)
  const route=/\bib\b|pyp|myp|dp/.test(q)?'ib':/英系|英国|剑桥|cambridge|ks[12]/.test(q)?'british':/香港|本地|dse/.test(q)?'hongkong':null
  const subject=/数学|英数|math/.test(q)?'math':/英语|英文|english/.test(q)?'english':/中文|语文|chinese/.test(q)?'chinese':null
  // A specific, uncurated series must reach the full catalog instead of being
  // swallowed by a broad stage recommendation (e.g. 小学启思语文新天地).
  const specific=q.replace(/香港|本地|英系|英国|剑桥|cambridge|dse|ib|pyp|小学|小[一二三四五六]|primary|ks[12]|数学|英数|英文数学|英语|英文|english|maths?|中文|语文|chinese|教材|资料|推荐|常规|进阶|拔尖|提高|advanced|授课|用哪个|用什么|请问|查找|搜索|找|的|有没有|一下|\s|[\d\p{P}]/gu,'')
  if(primary&&specific)return []
  if(primary&&!subject){
    return orderedEntries(route==='british'?['british-primary']:route==='ib'?['ib-pyp']:route==='hongkong'?['hk-primary']:['british-primary','ib-pyp','hk-primary'])
  }
  if(primary&&subject){
    const englishMedium=/英数|英文数学|英语授课|english.medium/.test(q)
    const mathIds=route==='british'?['oip-math']:route==='ib'?['ib-pyp-math','oip-math']:
      /进阶|拔尖|提高|advanced/.test(q)?['high-performance',englishMedium?'modern-en':'modern-cn']:
      englishMedium?['modern-en','high-performance']:['modern-cn','modern-en','high-performance']
    const ids=subject==='math'?mathIds:subject==='english'?(route==='british'?['global-english','oip-english']:route==='ib'?['oip-english','nelson']:['lighthouse','hk-primary']):['hk-primary']
    return orderedEntries(ids)
  }
  return []
}
