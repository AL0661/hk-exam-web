// Pendulum around the held scruff. Angle is unbounded; no fixed spin animation.
export function createDragInertia(){return {angle:0,omega:0,vx:0,vy:0,panic:0}}
export function moveDragInertia(state,dx,dy,seconds){
  const dt=Math.max(.008,Math.min(.1,seconds))
  const vx=Math.max(-2200,Math.min(2200,dx/dt)),vy=Math.max(-2200,Math.min(2200,dy/dt))
  // Pivot acceleration changes angular momentum; vertical motion matters when tilted.
  state.omega+=((vx-state.vx)*Math.cos(state.angle)+(vy-state.vy)*Math.sin(state.angle))/95
  state.omega=Math.max(-22,Math.min(22,state.omega))
  state.vx=vx;state.vy=vy
}
export function stepDragInertia(state,seconds){
  let remaining=Math.min(.1,Math.max(0,seconds))
  while(remaining>0){
    const dt=Math.min(1/120,remaining)
    state.omega+=(-18*Math.sin(state.angle)-.7*state.omega)*dt
    state.angle+=state.omega*dt;remaining-=dt
  }
  const lean=Math.atan2(Math.sin(state.angle),Math.cos(state.angle))
  if(Math.abs(lean)>Math.PI/3||Math.abs(state.omega)>4.5)state.panic=.8
  else state.panic=Math.max(0,state.panic-seconds)
  return {degrees:state.angle*180/Math.PI,flustered:state.panic>0}
}
export function releaseDragInertia(state){
  return {degrees:Math.atan2(Math.sin(state.angle),Math.cos(state.angle))*180/Math.PI,velocity:Math.max(-400,Math.min(400,state.omega*180/Math.PI))}
}
